option('keyring_dir',
  type: 'string',
  value: '/etc/apg/trusted.d',
  description: 'Default directory for trusted signing keys (auto-adjusted to /usr/local/etc/apg/trusted.d on FreeBSD unless overridden)',
)

option('keys_dir',
  type: 'string',
  value: '/etc/apg/keys',
  description: 'Default directory for libsodium key files (auto-adjusted to /usr/local/etc/apg/keys on FreeBSD unless overridden)',
)

option('tmp_dir',
  type: 'string',
  value: '/tmp/apg',
  description: 'Temporary directory used during package extraction',
)

option('db_mapsize',
  type: 'integer',
  value: 10485760,
  description: 'LMDB environment map size in bytes (default: 10 MiB)',
)

option('gpgme',
  type: 'feature',
  value: 'enabled',
  description: 'Build the gpgme (OpenPGP) signing backend, alongside the always-built libsodium one',
)

option('fuzzing',
  type: 'feature',
  value: 'disabled',
  description: 'Build libFuzzer harnesses under fuzz/ (requires clang; not part of a normal build)',
)
